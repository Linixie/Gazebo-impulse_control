CMAKE_MINIMUM_REQUIRED(VERSION 3.10.2 FATAL_ERROR)

FIND_PACKAGE(gz-cmake3 REQUIRED)

PROJECT(ImpulseControl)

GZ_FIND_PACKAGE(gz-plugin2 REQUIRED COMPONENTS register)

GZ_FIND_PACKAGE(gz-sim8 REQUIRED)

ADD_LIBRARY(ImpulseControl SHARED ImpulseControl.cc)
SET_PROPERTY(TARGET ImpulseControl PROPERTY CMAKE_CXX_STANDARD 17)
TARGET_LINK_LIBRARIES(ImpulseControl
  PUBLIC gz-plugin2::gz-plugin2
  PUBLIC gz-sim8::gz-sim8
)
